      "must": [

          "multi_match": {

            "fields": ["headline", "short_description"]
          }
        } 
      ],
      "must_not": [

          "multi_match": {
            "query": "Barack",
            "fields": ["headline", "short_description"]
          }
        }
      ]
      
    }
  }  
}

    "query": {
        "bool": {
          "must" : 
           [{"match_phrase": {"headline" : "Michelle Obama"}}],
           "must_not" :

           }          
        }
  }
