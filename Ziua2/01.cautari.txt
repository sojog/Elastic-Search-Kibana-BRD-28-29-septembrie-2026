# Cautare dupa anumite criterii - marimea rezultatelor
POST vorbe/_search
{
    "query": {
        "match": {
          "propozitie": "Ana"
        }
    }
}

# Combinarea celor 2 optiuni
POST vorbe/_search
{
    "size":1,
    "query": {
        "match": {
          "propozitie": "Ana"
        }
    }
}

# Minificarea JSON-ului
POST vorbe/_search
{"size":1,"query":{"match":{"propozitie":"Ana"}}}


# In cautarea textului (nu a keyword) - nu este important lowercase/uppercase

        }
    }
}

# Cautare or dupa Ana sau Florina

# Cautare cu fuzziness - cuvinte care au scriere apropiata
POST vorbe/_search
{
    "query": {
        "match": {
           "propozitie": {
            "query" : "Florina",
            "fuzziness": 1
           }
        }
    }
}

# Cautare cu parametri encodati
POST vorbe/_search?q="Floriana"


# Cautare cu mai multe cuvinte (Operatorul OR)
POST vorbe/_search
{
    "query": {
        "match": {
          "propozitie": "are pere"
        }
    }
}

# Cautare cu match_phrase exact - Aici cautarea trebuie sa fie cu propozitia exacta
POST vorbe/_search
{
    "query": {
        "match_phrase": {
          "propozitie": "are pere"
        }
    }
}
